{{- /* A site setting that must be a whole number of 1 or more (or of "min" or more), read in one place for tilesPerPage (wall-pager.html), photosPerGallery (wall-tiles.html) and search.textLimit (the search index). Context: a map with "name" (the setting's name, for the message), "value" (what the site set; nil for unset), "default" (returned for unset; 0 is a value, not "unset", #69) and "min" (the smallest allowed value; 1 when left out). Returns the number as an int. Fails the build naming the setting when the value isn't a number, is a boolean (int would read true as 1), a float with a fractional part (int would cut 2.9 to 2) or too big for a whole number (1e30), or is below the minimum (#256, #322). A string is read as decimal: its leading zeros are dropped, so "010" is 10 and "08" is 8, where int would read octal (so 8 and an error) (#322). */ -}} {{- $v := .value -}} {{- $n := .default -}} {{- $min := 1 -}} {{- if isset . "min" }}{{ $min = .min }}{{ end -}} {{- $want := cond (eq $min 1) "1 or more" (printf "%d or more" $min) -}} {{- if ne $v nil -}} {{- $bad := false -}} {{- $type := printf "%T" $v -}} {{- if eq $type "bool" }}{{ $bad = true }} {{- else if eq $type "float64" }}{{ if or (ne $v (math.Floor $v)) (gt (math.Abs $v) 1e15) }}{{ $bad = true }}{{ end }} {{- else if eq $type "string" }}{{ if not (findRE `^\s*[+-]?[0-9]+\s*$` $v) }}{{ $bad = true }}{{ else }}{{ $v = replaceRE `^([+-]?)0+([0-9])` "${1}${2}" (strings.TrimSpace $v) }}{{ end }} {{- end -}} {{- with try (int $v) -}} {{- if or $bad .Err }}{{ errorf "%s must be a whole number, %s (got %q)" $.name $want (printf "%v" $.value) }}{{ else }}{{ $n = .Value }}{{ end -}} {{- end -}} {{- end -}} {{- if lt $n $min }}{{ errorf "%s must be %s (got %d)" .name $want $n }}{{ end -}} {{- return $n -}}

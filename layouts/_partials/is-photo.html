{{- /* Photo or article? A post is a PHOTO post when its categories include "photos" (case-insensitive) or its raw content contains the galleries shortcode. Everything else is an article. Returns a bool. `categories` may be a list or a single string. */ -}} {{- $raw := .Params.categories -}} {{- if and $raw (not (reflect.IsSlice $raw)) }}{{ $raw = slice $raw }}{{ end -}} {{- $cats := slice -}} {{- range $raw -}}{{- $cats = $cats | append (lower .) -}}{{- end -}} {{- $hasShortcode := gt (len (findRE `\{\{[<%]\s*galleries\b` .RawContent 1)) 0 -}} {{- return (or (in $cats "photos") $hasShortcode) -}}

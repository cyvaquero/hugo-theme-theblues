{{- /* Photo or article? A post is a PHOTO post when its categories include "photos" (case-insensitive) or its raw content contains the galleries shortcode ({{< >}} form, outside HTML comments, #17). Everything else is an article. Returns a bool. `categories` may be a list or a single string. Kept in the page's Store by memo.html (#7, #164). */ -}} {{- $key := "theblues-is-photo" -}} {{- $out := false -}} {{- $m := partial "memo.html" (dict "page" . "key" $key) -}} {{- if $m }}{{ $out = $m.val }}{{ else -}} {{- $raw := .Params.categories -}} {{- if and $raw (not (reflect.IsSlice $raw)) }}{{ $raw = slice $raw }}{{ end -}} {{- $cats := slice -}} {{- range $raw -}}{{- $cats = $cats | append (lower .) -}}{{- end -}} {{- $hasShortcode := gt (len (findRE `\{\{<\s*galleries\b` (replaceRE `(?s)` "" .RawContent) 1)) 0 -}} {{- $out = or (in $cats "photos") $hasShortcode -}} {{- $_ := partial "memo.html" (dict "page" . "key" $key "val" $out) -}} {{- end -}} {{- return $out -}}

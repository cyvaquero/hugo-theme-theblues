{{- /* Returns a CSS/JS resource from assets/. Production builds are minified and fingerprinted (comments stripped, content hash in the file name). Development builds (hugo server) return the readable, commented source unchanged. A script whose first line is the template comment "theblues:template" (wall.js, comments.js, gallery.js, hero.js and search.js, which hold {{ partial "js-text.html" … }}) is run as a template first, so the text it shows comes from the site's language (#192). The comment renders as nothing. Any other script is published as written, even if it contains "{{" in a comment or a string, and a site's own copy of one of these five scripts needs the same first line (#262). On a site with more than one language the file is written once per language, with the language in its name (wall.de.js). Usage: {{ with partialCached "asset.html" "css/main.css" "css/main.css" }}...{{ .RelPermalink }}...{{ end }} Returns nil when the file doesn't exist, so a site's own partials can use it for optional files. The theme's own callers fail the build in that case (an {{ else }} with errorf), so a missing theme asset can't silently drop its tag from every page (#83). */ -}} {{- $r := resources.Get . -}} {{- if and $r (strings.HasSuffix . ".js") (strings.HasPrefix $r.Content "{{- /* theblues:template */ -}}") -}} {{- $to := . -}} {{- if gt (len hugo.Sites) 1 }}{{ $to = replaceRE `\.js$` (printf ".%s.js" site.Language.Lang) . }}{{ end -}} {{- $r = $r | resources.ExecuteAsTemplate $to dict -}} {{- end -}} {{- /* A script that reads the language's text but lost its marker would go out with the template action unexpanded. */ -}} {{- if and $r (strings.HasSuffix . ".js") (not (strings.HasPrefix $r.Content "{{- /* theblues:template */ -}}")) (strings.Contains $r.Content `{{ partial "js-text.html"`) -}} {{- errorf "theblues: %s calls js-text.html but its first line isn't the template comment \"theblues:template\", so the call would be published unexpanded (#262)" . -}} {{- end -}} {{- if and $r (not hugo.IsDevelopment) -}} {{- $r = $r | minify | fingerprint -}} {{- end -}} {{- /* Reading .Content here runs the template, minify and fingerprint steps now, while this partial runs. Left lazy, they would first run when Hugo hashes the returned resource inside --templateMetricsHints, which holds the metrics lock while the template's own partial calls wait for it, and the build hangs (#243). */ -}} {{- with $r }}{{ $_ := .Content }}{{ end -}} {{- return $r -}}

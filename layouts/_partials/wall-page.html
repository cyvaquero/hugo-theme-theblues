{{- /* Builds the tile slice for the given posts, paginates it BY TILE (params.wall.tilesPerPage, default 24), and renders the controls, the wall and the pager. Context: a map with "page" (the list page, which owns the paginator) and "posts" (sorted newest first). Hugo's .Paginate only accepts a collection of pages, not a slice of arbitrary values. So the pager is driven by a stand-in collection with exactly one entry per tile (the home page repeated); the pager supplies the page count, page number and URLs, and the tile slice is cut with the same offsets. */ -}} {{- $size := 24 -}} {{- with site.Params.wall }}{{ with .tilesPerPage }}{{ $size = int . }}{{ end }}{{ end -}} {{- $tiles := partial "wall-tiles.html" .posts -}} {{- $stand := slice -}} {{- range $tiles }}{{ $stand = $stand | append site.Home }}{{ end -}} {{- $pag := .page.Paginate $stand $size -}} {{- $start := mul (sub $pag.PageNumber 1) $size -}} {{ partial "controls.html" .page }} {{ partial "wall.html" (dict "tiles" (first $size (after $start $tiles))) }} {{ partial "pagination.html" $pag }}

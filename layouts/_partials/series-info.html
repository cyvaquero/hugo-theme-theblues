{{- /* Position of a page in each of its series, in front matter order: returns a slice of maps with name, url, part (1-based), total and pages, empty when the page is not in a series. Pages are in Hugo's weight order: posts with a weight first, lowest first, then the rest; oldest first within each (#6). series-box.html renders every entry; the "Part N of M" chip on wall tiles (tile-article.html) shows the first. Usage: {{ $series := partialCached "series-info.html" $page $page.Path }} */ -}} {{- $page := . -}} {{- $out := slice -}} {{- range .GetTerms "series" -}} {{- $byDate := .RegularPages.ByDate -}} {{- $pages := (sort (where $byDate "Weight" "gt" 0) "Weight") | append (where $byDate "Weight" 0) -}} {{- $part := 0 -}} {{- range $i, $p := $pages -}}{{- if eq $p.RelPermalink $page.RelPermalink -}}{{- $part = add $i 1 -}}{{- end -}}{{- end -}} {{- if $part -}} {{- $out = $out | append (dict "name" .LinkTitle "url" .RelPermalink "part" $part "total" (len $pages) "pages" $pages) -}} {{- end -}} {{- end -}} {{- return $out -}}

{{- /* Position of a page in each of its series, in front matter order: returns a slice of maps with name, url, part (1-based), total and pages (oldest first), empty when the page is not in a series. series-box.html renders every entry; the "Part N of M" chip on wall tiles (tile-article.html) shows the first. Usage: {{ $series := partialCached "series-info.html" $page $page.Path }} */ -}} {{- $page := . -}} {{- $out := slice -}} {{- range .GetTerms "series" -}} {{- $pages := .RegularPages.ByDate -}} {{- $part := 0 -}} {{- range $i, $p := $pages -}}{{- if eq $p.RelPermalink $page.RelPermalink -}}{{- $part = add $i 1 -}}{{- end -}}{{- end -}} {{- if $part -}} {{- $out = $out | append (dict "name" .LinkTitle "url" .RelPermalink "part" $part "total" (len $pages) "pages" $pages) -}} {{- end -}} {{- end -}} {{- return $out -}}

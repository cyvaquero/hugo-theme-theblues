{{- /* Parses the gallery shortcodes out of a page's raw Markdown and returns a slice of photos, each a map with src (thumbnail), full (lightbox URL, see smug-full.html) and title. Source order is kept. It follows what Hugo renders (#17): - HTML comments are removed first, so a commented-out photo is not counted, shown on walls or used as a cover. - Only the {{< >}} form counts; {{% %}} renders nothing. That includes a {{% galleries %}} block: the gallery calls inside it are dropped too, so they are not shown on walls or used as a cover or share image (#71). - Arguments are parsed like Hugo's: named (src=, title=) or positional (the first is src), quoted with "…" (\" and \\ escapes) or `…`, so a } or > inside a quoted title and an escaped quote are read correctly. A production build also keeps the result in the page's Store, since partialCached alone recomputes it on a big site: Hugo's partial cache holds about a thousand entries (measured with Hugo 0.167), so per-page entries evict each other (#7). Not under hugo server, where a page's Store survives a template edit and would go stale. Usage: {{ $photos := partialCached "photos.html" $page $page.Path }} */ -}} {{- $key := "theblues-photos" -}} {{- $out := cond hugo.IsServer nil (.Store.Get $key) -}} {{- if eq $out nil -}} {{- $raw := replaceRE `(?s)` "" .RawContent -}} {{- $raw = replaceRE `(?s)\{\{%\s*galleries\b.*?\{\{%\s*/galleries\s*%\}\}` "" $raw -}} {{- $q := "\"(?:[^\"\\\\]|\\\\.)*\"|`[^`]*`" -}} {{- $out = slice -}} {{- range findRESubmatch (printf "\\{\\{<\\s*gallery\\s((?:%s|[^\"`>])*)>\\}\\}" $q) $raw -}} {{- $named := dict -}} {{- $pos := slice -}} {{- range findRESubmatch (printf "(?:([A-Za-z_][A-Za-z0-9_]*)\\s*=\\s*)?(%s|[^\\s\"`]+)" $q) (index . 1) -}} {{- $v := index . 2 -}} {{- if hasPrefix $v "\"" }}{{ $v = $v | strings.TrimPrefix "\"" | strings.TrimSuffix "\"" | replaceRE `\\(.)` "$1" }} {{- else if hasPrefix $v "`" }}{{ $v = $v | strings.TrimPrefix "`" | strings.TrimSuffix "`" }}{{ end -}} {{- with index . 1 }}{{ $named = merge $named (dict (lower .) $v) }}{{ else }}{{ $pos = $pos | append $v }}{{ end -}} {{- end -}} {{- $src := $named.src | default (index $pos 0) | default "" -}} {{- $title := $named.title | default "" -}} {{- if $src -}} {{- $out = $out | append (dict "src" $src "full" (partialCached "smug-full.html" $src $src) "title" $title) -}} {{- end -}} {{- end -}} {{- if not hugo.IsServer }}{{ .Store.Set $key $out }}{{ end -}} {{- end -}} {{- return $out -}}

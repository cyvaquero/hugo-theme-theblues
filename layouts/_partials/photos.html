{{- /* Parses the gallery shortcodes out of a page's raw Markdown and returns a slice of photos, each a map with src (thumbnail), full (lightbox URL, see smug-full.html) and title. Source order is kept. It follows what Hugo renders (#17): - HTML comments are removed first, so a commented-out photo is not counted, shown on walls or used as a cover. - Only the {{< >}} form counts; {{% %}} renders nothing. That includes a {{% galleries %}} block: the gallery calls inside it are dropped too, so they are not shown on walls or used as a cover or share image (#71). - Arguments are parsed like Hugo's: named (src=, title=) or positional (the first is src), quoted with "…" (\" and \\ escapes) or `…`, so a } or > inside a quoted title and an escaped quote are read correctly. The result is also kept in the page's Store by memo.html, since partialCached alone recomputes it on a big site: Hugo's partial cache holds about a thousand entries (measured with Hugo 0.167), so per-page entries evict each other (#7). memo.html uses a kept result only within the build that kept it, so a rebuild under hugo server or hugo --watch never shows an old one (#164). Usage: {{ $photos := partialCached "photos.html" $page $page.Path }} */ -}} {{- $key := "theblues-photos" -}} {{- $out := slice -}} {{- $m := partial "memo.html" (dict "page" . "key" $key) -}} {{- if $m }}{{ $out = $m.val }}{{ else -}} {{- $raw := replaceRE `(?s)` "" .RawContent -}} {{- $raw = replaceRE `(?s)\{\{%\s*galleries\b.*?\{\{%\s*/galleries\s*%\}\}` "" $raw -}} {{- $q := "\"(?:[^\"\\\\]|\\\\.)*\"|`[^`]*`" -}} {{- $out = slice -}} {{- range findRESubmatch (printf "\\{\\{<\\s*gallery\\s((?:%s|[^\"`>])*)>\\}\\}" $q) $raw -}} {{- $named := dict -}} {{- $pos := slice -}} {{- range findRESubmatch (printf "(?:([A-Za-z_][A-Za-z0-9_]*)\\s*=\\s*)?(%s|[^\\s\"`]+)" $q) (index . 1) -}} {{- $v := index . 2 -}} {{- if hasPrefix $v "\"" }}{{ $v = $v | strings.TrimPrefix "\"" | strings.TrimSuffix "\"" | replaceRE `\\(.)` "$1" }} {{- else if hasPrefix $v "`" }}{{ $v = $v | strings.TrimPrefix "`" | strings.TrimSuffix "`" }}{{ end -}} {{- with index . 1 }}{{ $named = merge $named (dict (lower .) $v) }}{{ else }}{{ $pos = $pos | append $v }}{{ end -}} {{- end -}} {{- $src := $named.src | default (index $pos 0) | default "" -}} {{- $title := $named.title | default "" -}} {{- if $src -}} {{- $out = $out | append (dict "src" $src "full" (partialCached "smug-full.html" $src $src) "title" $title) -}} {{- end -}} {{- end -}} {{- $_ := partial "memo.html" (dict "page" . "key" $key "val" $out) -}} {{- end -}} {{- return $out -}}

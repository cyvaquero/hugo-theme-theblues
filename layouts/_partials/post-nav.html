{{- /* Previous (older) and next (newer) post by date, across all posts. */ -}} {{- $page := . -}} {{- $all := (where site.RegularPages "Section" "post").ByDate -}} {{- $idx := -1 -}} {{- range $i, $p := $all -}}{{- if eq $p.RelPermalink $page.RelPermalink -}}{{- $idx = $i -}}{{- end -}}{{- end -}} {{- if ge $idx 0 -}} {{- $older := false -}} {{- $newer := false -}} {{- if gt $idx 0 -}}{{- $older = index $all (sub $idx 1) -}}{{- end -}} {{- if lt (add $idx 1) (len $all) -}}{{- $newer = index $all (add $idx 1) -}}{{- end -}} {{- if or $older $newer }}
{{ with $older }}← {{ .Title }}{{ else }}{{ end }} {{ with $newer }}{{ .Title }} →{{ end }}
{{- end -}} {{- end -}}

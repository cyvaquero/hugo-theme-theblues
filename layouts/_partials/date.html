{{- /* A page date as shown on the site, or "Undated" when the page has none (#36). Context: a map with - date: the date (for example .Date); - format: a Go layout, default the site's params.dateFormat, else "Jan 2, 2006"; - time: true wraps the date in ; - undated: the text for a missing date, default "Undated" ("" for none). Returns a string, or HTML when time is set. assets/js/wall.js copies a gallery's date from its first tile's caption for its group headings, so they follow params.dateFormat too. Usage: {{ partial "date.html" (dict "date" .Date "time" true) }} */ -}} {{- $d := .date -}} {{- $out := "" -}} {{- if $d.IsZero -}} {{- $out = cond (isset . "undated") .undated "Undated" -}} {{- else -}} {{- $out = $d.Format (.format | default site.Params.dateFormat | default "Jan 2, 2006") -}} {{- if .time }}{{ $out = printf "%s" ($d.Format "2006-01-02") $out | safeHTML }}{{ end -}} {{- end -}} {{- return $out -}}

{{- /* An iframe embed, meant for SmugMug gallery embeds. Positional: URL, width, height and an optional title (default "Embedded content"; Hugo does not allow mixing positional and named arguments). The URL must be https and on another host than the site: the sandbox grants allow-scripts and allow-same-origin, which together let a same-origin page lift its own sandbox, so it only restricts cross-origin frames (no top navigation, forms or modals; #37). The frame sends no referrer beyond the origin. */ -}} {{- $src := .Get 0 -}} {{- $w := .Get 1 -}} {{- $h := .Get 2 -}} {{- if not (and $src $w $h) }}{{ errorf "embed: URL, width and height are required (%s)" .Position }}{{ end -}} {{- if not (strings.HasPrefix $src "https://") }}{{ errorf "embed: the URL must start with https:// (%s)" .Position }}{{ end -}} {{- if eq (lower (urls.Parse $src).Hostname) (lower (urls.Parse site.BaseURL).Hostname) }}{{ errorf "embed: the URL is on this site's own host, where the sandbox gives no protection (%s)" .Position }}{{ end -}}

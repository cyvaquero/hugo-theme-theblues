{{- /* An iframe embed, meant for SmugMug gallery embeds. Positional: URL, width, height and an optional title (default "Embedded content"; Hugo does not allow mixing positional and named arguments). The URL must be https and on another host than the site: the sandbox grants allow-scripts and allow-same-origin, which together let a same-origin page lift its own sandbox, so it only restricts cross-origin frames (no top navigation, forms or modals; #37). The frame sends no referrer beyond the origin. */ -}} {{- $src := .Get 0 -}} {{- $w := .Get 1 -}} {{- $h := .Get 2 -}} {{- if not (and $src $w $h) }}{{ errorf "embed: URL, width and height are required (%s)" .Position }}{{ end -}} {{- if not (strings.HasPrefix $src "https://") }}{{ errorf "embed: the URL must start with https:// (%s)" .Position }}{{ end -}} {{- /* Browsers and Go's URL parser disagree on some forms: https:///host/ has an empty host for Go but "host" for a browser, and a backslash counts as a slash for browsers. So the host must follow https:// directly, without userinfo, backslashes or spaces, and both hosts must be known before they are compared (#70). */ -}} {{- if not (findRE `^https://[^/\\?#@\s]+(/|$)` $src 1) }}{{ errorf "embed: the URL must be https://host/... with no user name, backslash or space before the path (%s)" .Position }}{{ end -}} {{- $host := lower (urls.Parse $src).Hostname -}} {{- $own := lower (urls.Parse site.BaseURL).Hostname -}} {{- if not $host }}{{ errorf "embed: the URL has no host (%s)" .Position }}{{ end -}} {{- if not $own }}{{ errorf "embed: the site's baseURL has no host, so the URL can't be checked against it; set baseURL to a full URL (%s)" .Position }}{{ end -}} {{- if eq $host $own }}{{ errorf "embed: the URL is on this site's own host, where the sandbox gives no protection (%s)" .Position }}{{ end -}}

{{- /* Markdown links. Only http, https, mailto and tel URLs and scheme-less paths are linked; anything else (javascript:, data:, vbscript:, or a URL that doesn't parse) renders the text without an href, as Goldmark does by default. A path that names a page or a page resource resolves to its permalink, like Hugo's built-in hook. Links to other hosts open in a new tab, and say so to screen readers. */ -}} {{- $href := "" -}} {{- $external := false -}} {{- $r := try (urls.Parse .Destination) -}} {{- if not $r.Err -}} {{- $u := $r.Value -}} {{- if in (slice "" "http" "https" "mailto" "tel") (lower $u.Scheme) -}} {{- $href = .Destination -}} {{- if and (not $u.Scheme) (not $u.Host) -}} {{- if strings.HasPrefix .Destination "#" -}} {{- $href = printf "%s%s" .PageInner.RelPermalink .Destination -}} {{- else if $u.Path -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- with or (.PageInner.GetPage $path) (.PageInner.Resources.Get $path) -}} {{- $href = .RelPermalink -}} {{- with $u.RawQuery }}{{ $href = printf "%s?%s" $href . }}{{ end -}} {{- with $u.Fragment }}{{ $href = printf "%s#%s" $href . }}{{ end -}} {{- end -}} {{- end -}} {{- else if $u.Host -}} {{- $external = ne (lower $u.Hostname) (lower (urls.Parse site.BaseURL).Hostname) -}} {{- end -}} {{- end -}} {{- end -}} {{ .Text }}{{ if $external }} {{ T "opensInNewTab" }}{{ end }} {{- /* No trailing newline: it would add a space after every link. */ -}}
